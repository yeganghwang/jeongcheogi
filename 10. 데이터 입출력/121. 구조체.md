# 1. 개요
> 자료의 종류가 다른 변수의 모임이다.

``` C
struct siljiju_student {
    char name[10];
    char grade[5];
    int private_point;
}

// 구조체 변수 선언하기
struct siljiju_student horikita, *kodo; // 구조체의 변수 이름은 'horikita' 이고 포인터는 kodo 이다.
horikita.name = "suzune";
horikita.grade = "1_D";
horikita.private_point: 12345;

// arrow 연산자로도 지정할 수 있다.
kodo->name = "suzune";

// 포인터처럼 사용할 수 있다.
(*kodo).name = "suzune";
(*kodo).private_point = 50000;

```
