# 1. 제어문의 개요
> 서술된 순서에 따라 무조건 위에서 아래로 실행되는 프로그램의 순서를 변경할 때 사용
``` C
// C언어
#include <stdio.h>
int main(void) {
    int score[] = {86, 53, 95, 76, 61};
    char grade;
    char str[] = "Rank";
    for (int i=0; i<5; i++) {
        switch (score[i] / 10) {
            case 10:
            case 9:
                grade = 'A';
                break;
            case 8:
                grade = 'B';
                break;
            case 7:
                grade = 'C';
                break;
            default: grade = 'F';
        }
        if (grade != 'F')
            printf("%d is %d %s\n", i+1, grade, str);
    }
}
```
